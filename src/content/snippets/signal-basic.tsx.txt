import { createSignal } from "solid-js";

const [count, setCount] = createSignal(0);

count();                       // 0，调用 getter 读取
setCount(1);                   // 直接写入
setCount((prev) => prev + 1);  // 基于前值更新

const [todos, setTodos] = createSignal<string[]>([]);
setTodos((list) => [...list, "读文档"]); // 对象/数组请整体替换
