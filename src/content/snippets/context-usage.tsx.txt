import {
  createContext,
  createStore,
  For,
  useContext,
  type ParentProps,
  type Store,
} from "solid-js";

type Todo = { id: number; title: string; done: boolean };
type TodoActions = { add(title: string): void; toggle(id: number): void };
type TodosCtx = readonly [Store<Todo[]>, TodoActions];

const TodosContext = createContext<TodosCtx>();

// 状态在组件里创建：SSR 下每个请求各有一份，Provider 负责拥有和销毁
// 模块作用域只留给常量（和 createContext 这一行本身）
export function TodosProvider(props: ParentProps) {
  const [todos, setTodos] = createStore<Todo[]>([]);

  const value: TodosCtx = [
    todos,
    {
      add: (title) =>
        setTodos((s) => {
          s.push({ id: s.length + 1, title, done: false });
        }),
      toggle: (id) =>
        setTodos((s) => {
          for (const todo of s) if (todo.id === id) todo.done = !todo.done;
        }),
    },
  ];

  return <TodosContext value={value}>{props.children}</TodosContext>;
}

// Provider 可以嵌套，最近的生效；后代里直接 useContext，拿到的就是 TodosCtx
function TodoList() {
  const [todos, { add, toggle }] = useContext(TodosContext);

  return (
    <>
      <button onClick={() => add("读 store 文档")}>新增</button>
      <ul>
        <For each={todos}>
          {(todo) => (
            <li>
              <button onClick={() => toggle(todo.id)}>
                {todo.done ? "完成" : "未完成"}
              </button>
              {todo.title}
            </li>
          )}
        </For>
      </ul>
    </>
  );
}
