import { createStore } from "solid-js";

type Profile = { name: string; nickname?: string; visits: number };

const [state, setState] = createStore<Profile>({
  name: "林岚",
  nickname: "岚岚",
  visits: 128,
});

// 读取就是普通属性访问：JSX 里写 {state.name}，只订阅 name 这一条路径
// 写入必须交给 setter，参数是一个可以随意改写的草稿
setState((s) => {
  s.visits += 1;
  s.nickname = "小岚";
});

// 删除一个键（1.x 写法：setState("nickname", undefined)）
setState((s) => {
  delete s.nickname;
});
