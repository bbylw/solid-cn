import { createStore } from "solid-js";

const [state, setState] = createStore({
  draft: { title: "", tags: [] as string[] },
  saving: false,
});

// 1.x：setState(produce((s) => { ... }))
// 2.0：produce 已经不存在，草稿写入就是 setter 的默认形式
setState((s) => {
  s.draft.title = "细粒度响应式";
  s.draft.tags.push("solid", "signals");
  s.saving = true;
});
