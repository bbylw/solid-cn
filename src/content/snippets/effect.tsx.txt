import { createEffect, createSignal } from "solid-js";

const [query, setQuery] = createSignal("");

createEffect(
  // compute：只做读取，依赖在这里登记
  () => query(),
  // apply：做副作用，可以返回清理函数；这里的读取不建立订阅
  (value) => console.log("当前查询词：", value)
);

// 写入排队，apply 在 compute 之后、这一帧的更新提交时运行
setQuery("signals"); // 输出：当前查询词：signals
