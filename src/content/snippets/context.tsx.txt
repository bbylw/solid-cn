import {
  createContext,
  createSignal,
  useContext,
  type Accessor,
  type ParentProps,
} from "solid-js";

type Theme = "light" | "dark";

// 无默认值：<T> 就是载荷本身，useContext 的返回类型是 Accessor<Theme>
// 没有 Provider 时运行时抛 ContextNotFoundError
const ThemeContext = createContext<Accessor<Theme>>();

export function ThemeProvider(props: ParentProps) {
  const [theme, setTheme] = createSignal<Theme>("light");

  // 2.0：context 自己就是 Provider，写 <ThemeContext value={...}>
  // 传的是 getter（theme），不是 theme()
  return (
    <>
      <button onClick={() => setTheme((t) => (t === "light" ? "dark" : "light"))}>
        切换主题
      </button>
      <ThemeContext value={theme}>{props.children}</ThemeContext>
    </>
  );
}

// 消费方直接 useContext，不需要判空包装的 useTheme
function ThemeBadge() {
  const theme = useContext(ThemeContext);
  return <span>当前主题：{theme()}</span>;
}
