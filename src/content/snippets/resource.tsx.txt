import { createMemo, createStore, Loading } from "solid-js";

type User = { id: number; name: string; email: string };

async function fetchUser(id: number): Promise<User> {
  const res = await fetch(`/api/users/${id}`);
  if (!res.ok) throw new Error("加载失败");
  return res.json();
}

// 1.x：const [user] = createResource(() => props.id, fetchUser);
// 2.0：异步只是计算返回值的一部分，读取方式与普通 memo 完全一致
function UserProfile(props: { id: number }) {
  const user = createMemo(() => fetchUser(props.id));

  return (
    <Loading fallback={<UserSkeleton />}>
      <h2>{user().name}</h2>
      <p>{user().email}</p>
    </Loading>
  );
}
