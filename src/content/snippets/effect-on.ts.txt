import { createEffect, createSignal, untrack } from "solid-js";

const [userId, setUserId] = createSignal(1);
const [token] = createSignal("local-token");

// 1.x 的 on(userId, fn)：2.0 里 compute 就是那份显式的依赖声明
createEffect(
  () => userId(),
  (id) => {
    // 只想拿当前值、不订阅的来源，用 untrack 包起来
    console.log("加载用户", id, "凭证", untrack(token));
  }
);

// 多个来源：在 compute 里组装成一个值
createEffect(
  () => [userId(), token()],
  ([id, tk]) => console.log(id, tk)
);

// 跳过首次执行：defer 挪到了第三个参数
createEffect(
  () => userId(),
  (id) => console.log("变为", id),
  { defer: true }
);

// compute 的第一个参数是上一轮的结果（首次为 undefined），用默认值给起点
createEffect(
  (prev = 0) => userId(),
  (value, prev) => console.log(`从 ${prev} 变成 ${value}`)
);

setUserId(2); // 写入仍然排队；提交之后，上面这些 effect 才一起跑
